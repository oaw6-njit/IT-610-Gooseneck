91## @brief Create accounts with default passwords (WILL REPLACE WITH DB AND HASHES LATER)

93RUN echo "root:rootpass" | chpasswd

95RUN useradd -m -s /bin/bash admin && \
96 echo "admin:adminpass" | chpasswd && \
97 usermod -aG sudo admin
98
99RUN useradd -m -s /bin/bash student && \
100 echo "student:studentpass" | chpasswd
101
102# Give student ownership of website content (optional)
103RUN mkdir -p /var/www/html && \
104chown -R student:student /var/www/html
105
106# Configure sudo permissions
107RUN cat > /etc/sudoers.d/lab-users <<'EOF'
108Cmnd_Alias NGINX_CMDS = \
109/etc/init.d/nginx start, \
110/etc/init.d/nginx stop, \
111/etc/init.d/nginx restart, \
112/etc/init.d/nginx reload, \
113/etc/init.d/nginx status
114
115admin ALL=(ALL:ALL) ALL
116student ALL=(root) NGINX_CMDS
117EOF
118
119RUN chmod 440 /etc/sudoers.d/lab-users
